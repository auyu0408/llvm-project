using namespace llvm;

namespace llvm{
    struct InstructionGraph{
        std::unordered_map<Instruction *, std::vector<Instruction *>> adjList;

        InstructionGraph(std::unordered_map<Instruction *, std::vector<Instruction *>>& G)
         : adjList(G) {}
    };
    
    template<> struct GraphTraits<InstructionGraph>{
        using NodeRef = Instruction *;
        using ChildIteratorType = std::vector<Instruction *>::const_iterator;

        //不管是 getEntryNode, child_begin 和 child_end 都需要注意 map 到空 instruction 的部份（可能是尾端）
        static Instruction* getEntryNode(const InstructionGraph &G) { 
            if(G.adjList.empty()) return nullptr;
            else return &(G.adjList.begin()->first->getFunction()->getEntryBlock().front()); 
        }
        static ChildIteratorType child_begin(NodeRef N) { 
            auto it = G->adjList.find(N);
            if(it == G->adjList.end()){
                static const std::vector<Instruction *> empty_vec;
                return empty_vec.begin();
            }
            else return it->second.begin(); 
        }
        static ChildIteratorType child_end(NodeRef N) { 
            auto it = G->adjList.find(N);
            if(it == G->adjList.end()){
                static const std::vector<Instruction *> empty_vec;
                return empty_vec.begin();
            }
            else return it->second.end();
        }

        static auto nodes_begin(const InstructionGraph &G) { return G.adjList.begin(); }
        static auto nodes_end(const InstructionGraph &G) { return G.adjList.end(); }

        static const InstructionGraph* G;
    };

    const InstructionGraph* GraphTraits<InstructionGraph>::G = nullptr;

}//namespace llvm

std::unordered_map<Instruction *, std::vector<Instruction *>> buildGraph(Function &F){
    std::unordered_map<Instruction *, std::vector<Instruction *>> G;
    G.clear();

     //處理指令
     for(BasicBlock &BB : F){
        Instruction *prevInst = nullptr;
        std::vector<Instruction *> Successors;
        
        for(Instruction &I : BB){
            if(prevInst){
                G[prevInst].push_back(&I);
            }
            prevInst = &I;
        }

        if(Instruction *TInst = BB.getTerminator()){
            for(unsigned i=0; i < TInst -> getNumSuccessors(); i++){
                BasicBlock *SuccBB = TInst -> getSuccessor(i);
                if(!SuccBB -> empty()){
                    G[TInst].push_back(&SuccBB->front());
                }
            }
        }
    }

    return G;
}

PreservedAnalyses HelloPass::run(Function &F, FunctionAnalysisManager &AM){
    std::unordered_map<Instruction *, std::vector<Instruction *>> CFG = buildGraph(F);
    InstructionGraph G(CFG);

    GraphTraits<InstructionGraph>::G = &G;
    
    if(CFG.empty()){
        errs() << "CFG empty" << "\n";
        return PreservedAnalyses::all();
    }
    
    int SCCIndex = 0;
    
    errs() << F.getName() << "\n";
    std::vector<std::vector<Instruction *>> SCCs;
    for(auto I = scc_begin(G); I != scc_end(G); ++I){
        std::vector<Instruction *> SCC = *I;

        // 只輸出真正的循環（大小 > 1 或自循環）
        if (SCC.size() > 1 || 
            (SCC.size() == 1 && CFG[SCC[0]].size() > 0 && CFG[SCC[0]][0] == SCC[0])) {
            SCCs.push_back(SCC);
        }
    }

    for(auto SCC:SCCs){
        errs() << "SCC #" << SCCIndex++ << ":\n";
        for (Instruction* Inst : SCC) {
            errs() << "  " << *Inst << "\n";
        }
    }


    return PreservedAnalyses::all();
}
